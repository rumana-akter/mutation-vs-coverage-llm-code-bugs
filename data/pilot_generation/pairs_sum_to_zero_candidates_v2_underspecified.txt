def pairs_sum_to_zero(l):
    seen = set()
    for x in l:
        if -x in seen:
            return True
        seen.add(x)
    return False

### IMPLEMENTATION 2 ###
def pairs_sum_to_zero(l):
    n = len(l)
    for i in range(n):
        for j in range(i + 1, n):
            if l[i] + l[j] == 0:
                return True
    return False

def pairs_sum_to_zero(l):
    s = sorted(l)
    left, right = 0, len(s) - 1
    while left < right:
        total = s[left] + s[right]
        if total == 0:
            return True
        elif total < 0:
            left += 1
        else:
            right -= 1
    return False

def pairs_sum_to_zero(l):
    from itertools import combinations
    return any(a + b == 0 for a, b in combinations(l, 2))

def pairs_sum_to_zero(l):
    counts = {}
    for x in l:
        counts[x] = counts.get(x, 0) + 1
    for x in counts:
        if x == 0:
            if counts[x] > 1:
                return True
        elif -x in counts:
            return True
    return False
