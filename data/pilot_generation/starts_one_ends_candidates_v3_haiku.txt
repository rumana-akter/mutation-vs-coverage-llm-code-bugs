def starts_one_ends(n):
    start_with_1 = 10**(n-1)
    if n == 1:
        end_with_1 = 1
        both = 1
    else:
        end_with_1 = 9 * 10**(n-2)
        both = 10**(n-2)
    return start_with_1 + end_with_1 - both

### IMPLEMENTATION 3 ###
def starts_one_ends(n):
    count = 0
    for num in range(10**(n-1), 10**n):
        s = str(num)
        if s[0] == '1' or s[-1] == '1':
            count += 1
    return count

def starts_one_ends(n):
    total_n_digit = 9 * 10**(n-1)
    if n == 1:
        return total_n_digit
    neither_start_nor_end_with_1 = 8 * 9 * 10**(n-2)
    return total_n_digit - neither_start_nor_end_with_1
