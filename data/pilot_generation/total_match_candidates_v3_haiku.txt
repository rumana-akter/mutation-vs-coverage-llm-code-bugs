def total_match(lst1, lst2):
    total1 = sum(len(s) for s in lst1)
    total2 = sum(len(s) for s in lst2)
    return lst1 if total1 <= total2 else lst2

def total_match(lst1, lst2):
    count1 = 0
    for s in lst1:
        count1 += len(s)
    count2 = 0
    for s in lst2:
        count2 += len(s)
    return lst1 if count1 <= count2 else lst2

def total_match(lst1, lst2):
    from functools import reduce
    total1 = reduce(lambda acc, s: acc + len(s), lst1, 0)
    total2 = reduce(lambda acc, s: acc + len(s), lst2, 0)
    return lst1 if total1 <= total2 else lst2

def total_match(lst1, lst2):
    total1 = sum(map(len, lst1))
    total2 = sum(map(len, lst2))
    return lst1 if total1 <= total2 else lst2

def total_match(lst1, lst2):
    char_count1 = len(''.join(lst1))
    char_count2 = len(''.join(lst2))
    return lst1 if char_count1 <= char_count2 else lst2
