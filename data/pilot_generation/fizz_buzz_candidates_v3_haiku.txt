def fizz_buzz(n: int):
    count = 0
    i = 0
    while i < n:
        if i % 11 == 0 or i % 13 == 0:
            for digit in str(i):
                if digit == '7':
                    count += 1
        i += 1
    return count

def fizz_buzz(n: int):
    def count_sevens(num):
        result = 0
        while num > 0:
            if num % 10 == 7:
                result += 1
            num //= 10
        return result

    total = 0
    for i in range(n):
        if i % 11 == 0 or i % 13 == 0:
            total += count_sevens(i)
    return total
