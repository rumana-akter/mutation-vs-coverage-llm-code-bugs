### IMPLEMENTATION 1 ###
def search(lst):
    from collections import Counter
    freq = Counter(lst)
    valid = [num for num, count in freq.items() if count >= num]
    return max(valid) if valid else -1

### IMPLEMENTATION 2 ###
def search(lst):
    freq = {}
    for num in lst:
        freq[num] = freq.get(num, 0) + 1

    result = -1
    for num, count in freq.items():
        if count >= num:
            result = max(result, num)
    return result

def search(lst):
    freq = {}
    for num in lst:
        freq[num] = freq.get(num, 0) + 1

    for num in sorted(freq.keys(), reverse=True):
        if freq[num] >= num:
            return num
    return -1

def search(lst):
    freq = {}
    for num in lst:
        freq[num] = freq.get(num, 0) + 1

    candidates = list(filter(lambda x: freq[x] >= x, freq.keys()))
    return max(candidates) if candidates else -1

### IMPLEMENTATION 5 ###
def search(lst):
    freq = {}
    for num in lst:
        freq[num] = freq.get(num, 0) + 1

    return max(
        (num for num, count in freq.items() if count >= num),
        default=-1
    )
