def decode_cyclic(s: str):
    groups = [s[(3 * i):min((3 * i + 3), len(s))] for i in range((len(s) + 2) // 3)]
    groups = [(group[-1] + group[:-1]) if len(group) == 3 else group for group in groups]
    return "".join(groups)

def decode_cyclic(s: str):
    result = []
    for i in range(0, len(s), 3):
        chunk = s[i:i+3]
        if len(chunk) == 3:
            result.append(chunk[2] + chunk[0] + chunk[1])
        else:
            result.append(chunk)
    return "".join(result)

def decode_cyclic(s: str):
    decoded = []
    for i in range(len(s)):
        group_start = (i // 3) * 3
        pos_in_group = i % 3
        group_size = min(3, len(s) - group_start)
        if group_size == 3:
            decoded.append(s[group_start + (pos_in_group - 1) % 3])
        else:
            decoded.append(s[i])
    return "".join(decoded)

def decode_cyclic(s: str):
    result = []
    idx = 0
    while idx < len(s):
        end_idx = min(idx + 3, len(s))
        group = s[idx:end_idx]
        if len(group) == 3:
            result.append(group[2] + group[0] + group[1])
        else:
            result.append(group)
        idx = end_idx
    return "".join(result)

def decode_cyclic(s: str):
    groups = [s[i:i+3] for i in range(0, len(s), 3)]
    decode_fn = lambda grp: (grp[-1] + grp[:-1]) if len(grp) == 3 else grp
    return "".join(map(decode_fn, groups))
