def cycpattern_check(a, b):
    if not b:
        return True
    return any(b[i:] + b[:i] in a for i in range(len(b)))

from collections import deque

def cycpattern_check(a, b):
    if not b:
        return True
    d = deque(b)
    for _ in range(len(b)):
        if ''.join(d) in a:
            return True
        d.rotate(-1)
    return False

def cycpattern_check(a, b):
    if not b:
        return True
    rotation = b
    for _ in range(len(b)):
        if rotation in a:
            return True
        rotation = rotation[1:] + rotation[0]
    return False

from itertools import islice, cycle

def cycpattern_check(a, b):
    if not b:
        return True
    for i in range(len(b)):
        rotation = ''.join(islice(cycle(b), i, i + len(b)))
        if rotation in a:
            return True
    return False
