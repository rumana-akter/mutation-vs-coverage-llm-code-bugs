def prod_signs(arr):
    if not arr:
        return None

    magnitude_sum = sum(abs(x) for x in arr)
    sign_product = 1
    for num in arr:
        if num > 0:
            sign_product *= 1
        elif num < 0:
            sign_product *= -1
        else:
            sign_product *= 0

    return magnitude_sum * sign_product

def prod_signs(arr):

        return None

        if x > 0:
            return 1
        elif x < 0:
            return -1
        else:
            return 0

        if sign_product == 0:

def prod_signs(arr):
    if not arr:
        return None

    from functools import reduce

    magnitude_sum = sum(abs(x) for x in arr)
    signs = [1 if x > 0 else -1 if x < 0 else 0 for x in arr]
    sign_product = reduce(lambda a, b: a * b, signs, 1)

    return magnitude_sum * sign_product

def prod_signs(arr):
    if len(arr) == 0:
        return None

    mag_sum = sum(abs(v) for v in arr)
    sp = 1
    for v in arr:
        sp *= (1 if v > 0 else -1 if v < 0 else 0)

    return mag_sum * sp
