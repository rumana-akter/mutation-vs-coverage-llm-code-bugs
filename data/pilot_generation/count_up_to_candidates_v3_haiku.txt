def count_up_to(n):
    if n <= 2:
        return []

    primes = []
    for num in range(2, n):
        is_prime = True
        for divisor in range(2, int(num**0.5) + 1):
            if num % divisor == 0:
                is_prime = False
                break
        if is_prime:
            primes.append(num)

    return primes

### IMPLEMENTATION 2 ###
def count_up_to(n):
    if n <= 2:
        return []

    is_prime = [True] * n
    is_prime[0] = is_prime[1] = False

    for i in range(2, int(n**0.5) + 1):
        if is_prime[i]:
            for j in range(i * i, n, i):
                is_prime[j] = False

    return [num for num in range(n) if is_prime[num]]

def count_up_to(n):
    def is_prime(num):
        if num < 2:
            return False
        if num == 2:
            return True
        if num % 2 == 0:
            return False
        for divisor in range(3, int(num**0.5) + 1, 2):
            if num % divisor == 0:
                return False
        return True

    return [num for num in range(2, n) if is_prime(num)]

### IMPLEMENTATION 4 ###
def count_up_to(n):
    primes = []

    for candidate in range(2, n):
        is_prime = True
        for prime in primes:
            if prime * prime > candidate:
                break
            if candidate % prime == 0:
                is_prime = False
                break
        if is_prime:
            primes.append(candidate)

    return primes

def count_up_to(n):
    if n < 2:
        return []

    primes = [2]
    for num in range(3, n, 2):
        if all(num % p != 0 for p in primes if p * p <= num):
            primes.append(num)

    return primes
