### IMPLEMENTATION 1 ###
def can_arrange(arr):
    for i in range(len(arr) - 1, 0, -1):
        if arr[i] < arr[i - 1]:
            return i
    return -1

### IMPLEMENTATION 2 ###
def can_arrange(arr):
    result = -1
    for i in range(1, len(arr)):
        if arr[i] < arr[i - 1]:
            result = i
    return result

### IMPLEMENTATION 3 ###
def can_arrange(arr):
    violations = [i for i in range(1, len(arr)) if arr[i] < arr[i - 1]]
    return max(violations) if violations else -1

### IMPLEMENTATION 4 ###
def can_arrange(arr):
    for i in range(len(arr) - 1, 0, -1):
        if not (arr[i] >= arr[i - 1]):
            return i
    return -1

### IMPLEMENTATION 5 ###
def can_arrange(arr):
    def find_violation(idx):
        if idx <= 0:
            return -1
        return idx if arr[idx] < arr[idx - 1] else find_violation(idx - 1)
    return find_violation(len(arr) - 1)
