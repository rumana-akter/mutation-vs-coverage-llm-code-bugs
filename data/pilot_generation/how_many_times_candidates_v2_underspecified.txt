### IMPLEMENTATION 1 ###
def how_many_times(string: str, substring: str) -> int:
    if not substring:
        return 0
    count = 0
    for i in range(len(string) - len(substring) + 1):
        if string[i:i + len(substring)] == substring:
            count += 1
    return count

### IMPLEMENTATION 2 ###
def how_many_times(string: str, substring: str) -> int:
    if not substring or not string:
        return 0
    count = 0
    start = 0
    while True:
        idx = string.find(substring, start)
        if idx == -1:
            break
        count += 1
        start = idx + 1
    return count

import re


def how_many_times(string: str, substring: str) -> int:
    if not substring:
        return 0
    return len(re.findall('(?=' + re.escape(substring) + ')', string))

### IMPLEMENTATION 4 ###
def how_many_times(string: str, substring: str) -> int:
    n = len(substring)
    if n == 0 or len(string) < n:
        return 0
    total = 0
    i = 0
    while i <= len(string) - n:
        if string[i:i + n] == substring:
            total += 1
        i += 1
    return total

def how_many_times(string: str, substring: str) -> int:
    if not substring:
        return 0

    def helper(s, sub, acc):
        pos = s.find(sub)
        if pos == -1:
            return acc
        return helper(s[pos + 1:], sub, acc + 1)

    return helper(string, substring, 0)
